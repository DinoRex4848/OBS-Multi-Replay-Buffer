cmake_minimum_required(VERSION 3.24)
project(obs-multi-replay VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
option(BUILD_PLUGIN "Build the OBS dock module" ON)
option(BUILD_TESTING "Build replay tests and native integration probe" ON)
option(REPLAY_INTEGRATION_TEST "Enable isolated OBS UI test driver (never enable in distributed builds)" OFF)
set(FFMPEG_ROOT "" CACHE PATH "FFmpeg prefix: OBS's Windows obs-deps folder, or /app inside the Flatpak SDK")
find_package(Threads REQUIRED)
if(WIN32)
  find_package(libobs 32 REQUIRED CONFIG)
  find_package(obs-frontend-api REQUIRED CONFIG)
  set(REPLAY_OBS OBS::libobs)
  set(REPLAY_FRONTEND OBS::obs-frontend-api)
else()
  find_path(OBS_INCLUDE_DIR obs.h PATH_SUFFIXES obs REQUIRED)
  find_library(OBS_LIBRARY NAMES obs REQUIRED)
  add_library(replay_obs INTERFACE)
  target_include_directories(replay_obs SYSTEM INTERFACE "${OBS_INCLUDE_DIR}")
  target_link_libraries(replay_obs INTERFACE "${OBS_LIBRARY}")
  target_compile_definitions(replay_obs INTERFACE HAVE_OBSCONFIG_H)
  set(REPLAY_OBS replay_obs)
  find_library(REPLAY_FRONTEND NAMES obs-frontend-api REQUIRED)
endif()
find_path(FFMPEG_INCLUDE_DIR libavformat/avformat.h HINTS "${FFMPEG_ROOT}/include" PATH_SUFFIXES ffmpeg REQUIRED)
# Linux needs FFmpeg 8 (avformat/avcodec 62, avutil 60); keep in step with scripts/project.py.
set(REPLAY_FFMPEG_MAJOR_avformat 62)
set(REPLAY_FFMPEG_MAJOR_avcodec 62)
set(REPLAY_FFMPEG_MAJOR_avutil 60)
foreach(component avformat avcodec avutil)
  set(names ${component})
  if(NOT WIN32)
    list(APPEND names lib${component}.so.${REPLAY_FFMPEG_MAJOR_${component}})
  endif()
  find_library(${component}_LIBRARY NAMES ${names} HINTS "${FFMPEG_ROOT}/lib" REQUIRED)
endforeach()
if(UNIX AND NOT APPLE)
  include(CheckCXXSourceRuns)
  include(CMakePushCheckState)
  cmake_push_check_state(RESET)
  set(CMAKE_REQUIRED_INCLUDES "${FFMPEG_INCLUDE_DIR}")
  set(CMAKE_REQUIRED_LIBRARIES "${avformat_LIBRARY}" "${avcodec_LIBRARY}" "${avutil_LIBRARY}")
  # Recheck when a development prefix or its libraries change in an existing build.
  unset(REPLAY_FFMPEG8_MATCHES CACHE)
  check_cxx_source_runs("extern \"C\" {
    #include <libavformat/avformat.h>
    #include <libavcodec/avcodec.h>
    #include <libavutil/avutil.h>
    }
    static_assert(LIBAVFORMAT_VERSION_MAJOR == 62, \"FFmpeg 8 avformat headers required\");
    static_assert(LIBAVCODEC_VERSION_MAJOR == 62, \"FFmpeg 8 avcodec headers required\");
    static_assert(LIBAVUTIL_VERSION_MAJOR == 60, \"FFmpeg 8 avutil headers required\");
    int main() {
      return (avformat_version() >> 16) != 62 ||
             (avcodec_version() >> 16) != 62 ||
             (avutil_version() >> 16) != 60;
    }" REPLAY_FFMPEG8_MATCHES)
  cmake_pop_check_state()
  if(NOT REPLAY_FFMPEG8_MATCHES)
    message(FATAL_ERROR "Linux requires matching FFmpeg 8 headers and runtime libraries (avformat 62, avcodec 62, avutil 60). Use the development environment matching your OBS installation.")
  endif()
  message(STATUS "Verified FFmpeg 8 headers and runtime libraries: ${FFMPEG_INCLUDE_DIR}; ${avformat_LIBRARY}; ${avcodec_LIBRARY}; ${avutil_LIBRARY}")
endif()
# Plain C++ with no OBS or FFmpeg dependency, so tests run everywhere.
add_library(replay_core STATIC src/history.cpp src/naming.cpp src/status_text.cpp)
target_include_directories(replay_core PUBLIC src)
target_link_libraries(replay_core PUBLIC Threads::Threads)
add_library(replay_capture STATIC src/capture.cpp src/writer_process.cpp)
target_include_directories(replay_capture PUBLIC src "${FFMPEG_INCLUDE_DIR}")
target_link_libraries(replay_capture PUBLIC replay_core ${REPLAY_OBS} ${avformat_LIBRARY} ${avcodec_LIBRARY} ${avutil_LIBRARY})
# Profile settings JSON, shared by the plugin and its tests.
add_library(replay_preferences STATIC src/preferences.cpp)
target_include_directories(replay_preferences PUBLIC src)
target_link_libraries(replay_preferences PUBLIC ${REPLAY_OBS})
add_executable(multi-replay-writer src/writer.cpp)
target_include_directories(multi-replay-writer PRIVATE src "${FFMPEG_INCLUDE_DIR}")
target_link_libraries(multi-replay-writer PRIVATE ${avformat_LIBRARY} ${avcodec_LIBRARY} ${avutil_LIBRARY})
if(BUILD_TESTING)
  enable_testing()
  add_executable(history_tests tests/history_tests.cpp)
  target_link_libraries(history_tests PRIVATE replay_core)
  add_test(NAME history COMMAND history_tests)
  add_executable(naming_tests tests/naming_tests.cpp)
  target_link_libraries(naming_tests PRIVATE replay_core)
  add_test(NAME naming COMMAND naming_tests)
  add_executable(timebase_tests tests/timebase_tests.cpp)
  target_include_directories(timebase_tests PRIVATE src)
  add_test(NAME timebase COMMAND timebase_tests)
  add_executable(status_text_tests tests/status_text_tests.cpp)
  target_link_libraries(status_text_tests PRIVATE replay_core)
  add_test(NAME status_text COMMAND status_text_tests)
  add_executable(preferences_tests tests/preferences_tests.cpp)
  target_link_libraries(preferences_tests PRIVATE replay_preferences)
  # Needs a real OBS runtime and encoder; see docs/DEVELOPMENT.md.
  add_executable(obs_probe tests/obs_probe.cpp)
  target_link_libraries(obs_probe PRIVATE replay_capture)
  add_executable(fault-writer tests/fault_writer.cpp)
  target_link_libraries(fault-writer PRIVATE Threads::Threads)
  add_executable(mux_tests tests/mux_tests.cpp)
  target_link_libraries(mux_tests PRIVATE replay_capture)
  # On Windows, CTest runs without OBS's and FFmpeg's DLL folders on PATH; the
  # Windows package tests exercise the installed writer instead.
  if(NOT WIN32)
    add_test(NAME preferences COMMAND preferences_tests)
    # Remuxes a small committed AV1/AAC clip, then checks collisions, failures and a stalled writer.
    set(REPLAY_MUX_OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/mux-test-output")
    add_test(NAME mux_output_reset COMMAND "${CMAKE_COMMAND}" -E rm -rf "${REPLAY_MUX_OUTPUT}")
    add_test(NAME mux COMMAND mux_tests $<TARGET_FILE:multi-replay-writer> $<TARGET_FILE:fault-writer>
                              "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/mux-input.mkv" "${REPLAY_MUX_OUTPUT}")
    set_tests_properties(mux_output_reset PROPERTIES FIXTURES_SETUP mux_output)
    set_tests_properties(mux PROPERTIES FIXTURES_REQUIRED mux_output TIMEOUT 60)
  endif()
endif()
if(BUILD_PLUGIN)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
  add_library(obs-multi-replay MODULE
    src/module.cpp src/replay_controller.cpp src/replay_dock.cpp src/settings_dialog.cpp
    src/hotkeys.cpp src/recording_encoders.cpp)
  if(REPLAY_INTEGRATION_TEST)
    target_sources(obs-multi-replay PRIVATE src/integration_test.cpp)
    target_compile_definitions(obs-multi-replay PRIVATE REPLAY_INTEGRATION_TEST)
  endif()
  target_link_libraries(obs-multi-replay PRIVATE replay_capture replay_preferences ${REPLAY_FRONTEND} Qt6::Widgets)
  set_target_properties(obs-multi-replay PROPERTIES PREFIX "")
  if(WIN32)
    install(TARGETS obs-multi-replay RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
    install(DIRECTORY data/ DESTINATION data/obs-plugins/obs-multi-replay)
    install(TARGETS multi-replay-writer RUNTIME DESTINATION data/obs-plugins/obs-multi-replay)
    if(MSVC)
      # The helper runs outside OBS's executable folder. Bundle the compiler's
      # release CRT beside it so portable OBS does not need a system install.
      set(CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS_SKIP TRUE)
      include(InstallRequiredSystemLibraries)
      if(NOT CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS)
        message(FATAL_ERROR "MSVC redistributable runtime files were not found; required for the portable writer")
      endif()
      install(FILES ${CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS} DESTINATION data/obs-plugins/obs-multi-replay)
    endif()
  else()
    include(GNUInstallDirs)
    install(TARGETS obs-multi-replay LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/obs-plugins)
    install(DIRECTORY data/ DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/obs/obs-plugins/obs-multi-replay)
    install(TARGETS multi-replay-writer RUNTIME DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/obs/obs-plugins/obs-multi-replay)
  endif()
endif()
